function fn(text,name,tag){
 const start=text.toLowerCase().indexOf(`function ${name.toLowerCase()}(`)
 assert(start>=0,`Missing ${name}`)
 const line=text.lastIndexOf('\n',start)+1
 const end=text.indexOf(`${tag};`,text.indexOf(tag,start)+tag.length)
 assert(end>start,`Missing terminator ${name}`)
 return text.slice(line,end+tag.length+1)
}
